# Meson cross file for the iPod 5G/5.5G ARM target.
#
# The PortalPlayer PP5022 SoC has dual ARM7TDMI cores running at up to
# 80 MHz. We target ARMv4T (the ARM7TDMI ISA). Toolchain is the standard
# arm-none-eabi-gcc bundle.
#
# TOOLCHAIN VERSION. This comment used to say "gcc 14 on Arch's official
# package", and that had gone stale — the package had rolled forward to
# 16.1.0, gcc 15 having meanwhile added -Wunterminated-string-initialization,
# and the -Werror hw build stopped compiling. That is exactly the break the
# stale comment made harder to diagnose, so the pin now lives somewhere that
# is actually checked:
#
#   .tool-versions               the human-readable record and the reasoning
#   .github/workflows/ci.yml     ARM_TOOLCHAIN_VERSION — the pinned reference
#                                build (Arm GNU Toolchain 14.2.rel1), plus a
#                                second job on the rolling Arch toolchain so
#                                the next upgrade breaks CI, not a laptop
#
# Floor is gcc 14; there is deliberately no ceiling. core/meson.build probes
# for newer-compiler flags with cc.get_supported_arguments() instead of
# passing them blind, so both ends of that range build.
#
# Usage:
#   meson setup build-hw --cross-file cross/arm-none-eabi.txt -Dtarget=hw

[binaries]
c          = 'arm-none-eabi-gcc'
cpp        = 'arm-none-eabi-g++'
ar         = 'arm-none-eabi-ar'
strip      = 'arm-none-eabi-strip'
objcopy    = 'arm-none-eabi-objcopy'
ld         = 'arm-none-eabi-ld'
pkg-config = 'pkg-config'

[host_machine]
system     = 'none'
cpu_family = 'arm'
cpu        = 'arm7tdmi'
endian     = 'little'

[built-in options]
# Meson machine files use configparser syntax; multi-line list values
# must be indented continuation lines, so we keep these single-line.
c_args = ['-mcpu=arm7tdmi', '-ffreestanding']
c_link_args = ['-nostdlib', '-nostartfiles']
